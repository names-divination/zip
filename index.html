// ZIP化してダウンロードするメイン処理（0バイトファイル対応版） downloadBtn.addEventListener('click', async () => { downloadBtn.disabled = true; downloadBtn.innerText = 'ZIPを作成中...'; try { const zip = new JSZip(); // 選択されたファイルを一つずつZIPに追加（0バイトの空ファイルもそのまま保持） for (const file of selectedFiles) { zip.file(file.name, file); } // ZIPファイルの生成 const content = await zip.generateAsync({ type: 'blob' }); // ブラウザ標準のリンク機能でダウンロードを実行 const url = URL.createObjectURL(content); const a = document.createElement('a'); a.href = url; a.download = 'archive.zip'; document.body.appendChild(a); a.click(); // 疑似クリック // 後片付け document.body.removeChild(a); URL.revokeObjectURL(url); } catch (error) { alert('ZIP作成中にエラーが発生しました。開発者コンソールを確認してください。'); console.error(error); } finally { downloadBtn.disabled = false; downloadBtn.innerText = 'ZIPにしてダウンロード'; } });
